# Plan 5b-1: Juice — Effects Director and Animations — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make changes in the city feel impactful, like a game.
- **Venue opens:** a screen-space fisheye shockwave expands from the venue, the landmark springs up out of the ground with an overshoot, and the camera shakes a little.
- **Venue closes:** an implosion ripple, and the landmark sinks while its sign flickers out.
- **Events:** a pulsing light beacon with expanding ground rings.
- **New and split scenes:** a coloured ground ripple at the scene's heart.
- **Agents switching scene:** they pop and blend to their new colour rather than snapping.

**Architecture:**
- **Frames:** each frame gains an `epoch` (bumped on init and rewind) and the `sceneEvents` of the tick that produced it.
- **`diffFrames`:** a pure function that turns two consecutive same-epoch frames into a list of changes.
- **`EffectsDirector`:** a pure, time-injected class that turns changes into timed effects: waves, ripples and shake.
- **Rendering:**
  - a custom `postprocessing` `Effect` (`ShockwaveEffect`) distorts screen UVs, using up to 4 simultaneous waves plus a global shake offset;
  - a ground-ripple layer draws the coloured rings;
  - landmarks animate their own rise and sink with a critically damped spring;
  - crowds blend their colours over time.
- **Load time:** the city layout moves into a Web Worker, so loading London no longer freezes the page.

**Tech Stack:** as before (Three.js, React Three Fiber 9, `postprocessing` 6, `@react-three/postprocessing` 3, Vitest).

`src/render/effects/diff.ts`:
```ts
import type { Frame } from "@/sim/frame";

export type FrameChange =
  | { kind: "venueOpened"; slot: number }
  | { kind: "venueClosed"; slot: number }
  | { kind: "eventStarted"; slot: number }
  | { kind: "eventEnded"; slot: number }
  | { kind: "sceneBorn"; scene: number; parent: number | null }
  | { kind: "sceneEnded"; scene: number; into: number | null };

/**
 * What visibly changed between two frames. Empty unless `next` directly follows `prev` in the same
 * epoch, so rewinds, re-inits and gaps never trigger effects.
 */
export function diffFrames(prev: Frame | null, next: Frame): FrameChange[] {
  if (!prev || prev.epoch !== next.epoch || next.tick !== prev.tick + 1) return [];
  const changes: FrameChange[] = [];
  for (let v = 0; v < next.nVenues; v++) {
    const isEvent = next.venueExpires[v] >= 0;
    const known = v < prev.nVenues && prev.venueEntity[v] === next.venueEntity[v] && prev.venueCell[v] === next.venueCell[v];
    const wasOpen = known && prev.venueOpen[v] === 1;
    const isOpen = next.venueOpen[v] === 1;
    if (!wasOpen && isOpen) changes.push({ kind: isEvent ? "eventStarted" : "venueOpened", slot: v });
    else if (wasOpen && !isOpen) changes.push({ kind: isEvent ? "eventEnded" : "venueClosed", slot: v });
  }
  for (const e of next.sceneEvents) {
    if (e.kind === "birth") changes.push({ kind: "sceneBorn", scene: e.scene, parent: null });
    else if (e.kind === "split") changes.push({ kind: "sceneBorn", scene: e.scene, parent: e.other });
    else if (e.kind === "merge") changes.push({ kind: "sceneEnded", scene: e.scene, into: e.other });
    else changes.push({ kind: "sceneEnded", scene: e.scene, into: null });
  }
  return changes;
}
```

`src/render/effects/director.ts`:
```ts
import type { FrameChange } from "@/render/effects/diff";
import type { XZ } from "@/render/layout";
import type { Rgb } from "@/render/palette";

export const MAX_WAVES = 4;

/** Screen-space shockwave anchored at a world point. strength > 0 bulges out, < 0 implodes. */
export type Wave = { x: number; z: number; start: number; duration: number; strength: number };
/** Expanding coloured ring on the ground. */
export type Ripple = { x: number; z: number; start: number; duration: number; radius: number; rgb: Rgb };

export type Anchors = {
  venue(slot: number): XZ | null;
  scene(id: number): XZ | null;
  sceneRgb(id: number): Rgb;
  venueRgb(slot: number): Rgb;
};

const GREY: Rgb = [0.6, 0.62, 0.7];

/**
 * Turns frame changes into timed effects. Pure apart from its own state: time is passed in, so it
 * is testable and deterministic. Renderers read `waves`, `ripples` and `shake(now)` every frame.
 */
export class EffectsDirector {
  waves: Wave[] = [];
  ripples: Ripple[] = [];
  private shakeStart = -Infinity;
  private shakeDuration = 0;
  private shakeAmp = 0;

  ingest(changes: FrameChange[], now: number, anchors: Anchors): void {
    for (const c of changes) {
      if (c.kind === "venueOpened" || c.kind === "venueClosed" || c.kind === "eventStarted" || c.kind === "eventEnded") {
        const p = anchors.venue(c.slot);
        if (!p) continue;
        const opening = c.kind === "venueOpened" || c.kind === "eventStarted";
        if (c.kind !== "eventEnded") {
          this.addWave({ x: p[0], z: p[1], start: now, duration: opening ? 1200 : 900, strength: opening ? 1 : -0.7 });
          this.kick(now, opening ? 0.006 : 0.004, opening ? 380 : 280);
        }
        this.ripples.push({
          x: p[0], z: p[1], start: now,
          duration: opening ? 1400 : 900,
          radius: c.kind === "eventStarted" ? 60 : opening ? 30 : 18,
          rgb: opening ? anchors.venueRgb(c.slot) : GREY,
        });
      } else {
        const p = anchors.scene(c.scene);
        if (!p) continue;
        const born = c.kind === "sceneBorn";
        this.ripples.push({
          x: p[0], z: p[1], start: now,
          duration: born ? 1800 : 1200,
          radius: born ? 45 : 25,
          rgb: born ? anchors.sceneRgb(c.scene) : GREY,
        });
      }
    }
  }

  /** Wave/ripple progress in [0, 1]. */
  progress(effect: { start: number; duration: number }, now: number): number {
    return Math.max(0, Math.min(1, (now - effect.start) / effect.duration));
  }

  /** Screen-UV shake offset at `now`, decaying to [0, 0]. */
  shake(now: number): [number, number] {
    const t = (now - this.shakeStart) / this.shakeDuration;
    if (!(t >= 0 && t < 1)) return [0, 0];
    const amp = this.shakeAmp * (1 - t) * (1 - t);
    return [amp * Math.sin(now * 0.09), amp * Math.cos(now * 0.071)];
  }

  prune(now: number): void {
    this.waves = this.waves.filter((w) => now - w.start < w.duration);
    this.ripples = this.ripples.filter((r) => now - r.start < r.duration);
  }

  private addWave(w: Wave): void {
    this.waves.unshift(w);
    if (this.waves.length > MAX_WAVES) this.waves.length = MAX_WAVES;
  }

  private kick(now: number, amp: number, duration: number): void {
    const remaining = this.shake(now);
    if (Math.hypot(remaining[0], remaining[1]) > amp) return;
    this.shakeStart = now;
    this.shakeDuration = duration;
    this.shakeAmp = amp;
  }
}
```

```ts
import { BlendFunction, Effect } from "postprocessing";
import { Uniform, Vector2, Vector4 } from "three";
import { MAX_WAVES } from "@/render/effects/director";

const FRAGMENT = /* glsl */ `
  uniform vec4 uWaves[${MAX_WAVES}];
  uniform vec2 uShake;
  uniform float uAspect;

  // Fisheye bulge behind the front plus a sharp refractive ring at the front.
  // wave = (u, v, progress, strength); strength < 0 pulls inward (implosion).
  vec2 displacement(vec2 uv, vec4 wave) {
    if (wave.w == 0.0) return vec2(0.0);
    vec2 d = uv - wave.xy;
    d.x *= uAspect;
    float dist = length(d);
    float radius = wave.z * 0.55;
    float fade = 1.0 - wave.z;
    float ring = exp(-pow((dist - radius) / 0.025, 2.0));
    float bulge = (1.0 - smoothstep(0.0, max(radius, 0.001), dist)) * 0.6;
    vec2 dir = dist > 0.0001 ? d / dist : vec2(0.0);
    dir.x /= uAspect;
    return dir * (ring * 0.035 + bulge * 0.025) * fade * wave.w;
  }

  void mainUv(inout vec2 uv) {
    vec2 offset = uShake;
    for (int i = 0; i < ${MAX_WAVES}; i++) offset -= displacement(uv, uWaves[i]);
    uv += offset;
  }

  void mainImage(const in vec4 inputColor, const in vec2 uv, out vec4 outputColor) {
    vec3 glow = vec3(0.0);
    for (int i = 0; i < ${MAX_WAVES}; i++) {
      vec4 w = uWaves[i];
      if (w.w == 0.0) continue;
      vec2 d = uv - w.xy;
      d.x *= uAspect;
      float ring = exp(-pow((length(d) - w.z * 0.55) / 0.018, 2.0));
      glow += ring * (1.0 - w.z) * (w.w > 0.0 ? vec3(0.9, 0.85, 1.2) : vec3(0.5, 0.55, 0.8)) * abs(w.w) * 0.35;
    }
    outputColor = vec4(inputColor.rgb + glow, inputColor.a);
  }
`;

export type ScreenWave = { u: number; v: number; progress: number; strength: number };

/** Screen-space shockwaves (up to MAX_WAVES) with a refractive ring, a fisheye bulge and a faint glow, plus global shake. */
export class ShockwaveEffect extends Effect {
  constructor() {
    super("ShockwaveEffect", FRAGMENT, {
      blendFunction: BlendFunction.NORMAL,
      uniforms: new Map<string, Uniform>([
        ["uWaves", new Uniform(Array.from({ length: MAX_WAVES }, () => new Vector4(0, 0, 0, 0)))],
        ["uShake", new Uniform(new Vector2(0, 0))],
        ["uAspect", new Uniform(1)],
      ]),
    });
  }

  setWaves(waves: ScreenWave[]): void {
    const slots = this.uniforms.get("uWaves")?.value as Vector4[];
    for (let i = 0; i < slots.length; i++) {
      const w = waves[i];
      if (w) slots[i].set(w.u, w.v, w.progress, w.strength);
      else slots[i].set(0, 0, 0, 0);
    }
  }

  setShake(x: number, y: number): void {
    (this.uniforms.get("uShake")?.value as Vector2).set(x, y);
  }

  setAspect(aspect: number): void {
    const u = this.uniforms.get("uAspect");
    if (u) u.value = aspect;
  }
}

"use client";

import { useFrame, useThree } from "@react-three/fiber";
import { type RefObject, useLayoutEffect, useMemo, useRef } from "react";
import * as THREE from "three";
import { diffFrames } from "@/render/effects/diff";
import { EffectsDirector } from "@/render/effects/director";
import type { ShockwaveEffect } from "@/render/effects/ShockwaveEffect";
import type { FrameStore } from "@/render/frame-store";
import { agentHomePosition, type CityLayout } from "@/render/layout";
import { type Rgb, sceneHuesOf, sceneRgb } from "@/render/palette";
import { venueViewsOf } from "@/render/venues";
import type { Frame } from "@/sim/frame";
import type { World } from "@/world/schema";

const MAX_RIPPLES = 12;
const NEUTRAL: Rgb = [0.9, 0.85, 0.75];

/** Mean home position of a scene's members (computed only when a scene event needs an anchor). */
function sceneAnchor(frame: Frame, layout: CityLayout, scene: number): [number, number] | null {
  let x = 0;
  let z = 0;
  let n = 0;
  for (let i = 0; i < frame.scene.length; i++) {
    if (frame.scene[i] !== scene || !frame.alive[i]) continue;
    const [hx, hz] = agentHomePosition(layout, frame.homeCell[i], i);
    x += hx;
    z += hz;
    n++;
  }
  return n > 0 ? [x / n, z / n] : null;
}

/**
 * Runs the effects director: diffs each new frame, then feeds the shockwave effect (world → screen)
 * and draws ground ripples.
 */
export function Effects({
  world, layout, frames, shockwave,
}: {
  world: World;
  layout: CityLayout;
  frames: RefObject<FrameStore>;
  shockwave: ShockwaveEffect;
}) {
  const director = useRef(new EffectsDirector());
  const lastFrame = useRef<Frame | null>(null);
  const ripplesRef = useRef<THREE.InstancedMesh>(null);
  const camera = useThree((s) => s.camera);
  const size = useThree((s) => s.size);
  const scratch = useMemo(() => ({ v: new THREE.Vector3(), m: new THREE.Matrix4(), c: new THREE.Color() }), []);

  useLayoutEffect(() => {
    const mesh = ripplesRef.current;
    if (!mesh) return;
    const hidden = new THREE.Matrix4().makeScale(0, 0, 0);
    for (let i = 0; i < MAX_RIPPLES; i++) mesh.setMatrixAt(i, hidden);
    mesh.instanceMatrix.needsUpdate = true;
  }, []);

  useFrame(() => {
    const now = performance.now();
    const d = director.current;
    const frame = frames.current.current;
    if (frame && frame !== lastFrame.current) {
      const changes = diffFrames(lastFrame.current, frame);
      if (changes.length > 0) {
        const views = venueViewsOf(frame, layout, world);
        const hues = sceneHuesOf(frame.lineageParents);
        d.ingest(changes, now, {
          venue: (slot) => views[slot]?.position ?? null,
          scene: (id) => sceneAnchor(frame, layout, id),
          sceneRgb: (id) => sceneRgb(hues, id),
          venueRgb: (slot) => {
            const s = views[slot]?.dominantScene ?? -1;
            return s >= 0 ? sceneRgb(hues, s) : NEUTRAL;
          },
        });
      }
      lastFrame.current = frame;
    }
    d.prune(now);

    shockwave.setAspect(size.width / Math.max(1, size.height));
    shockwave.setWaves(
      d.waves.map((w) => {
        const p = scratch.v.set(w.x, 0, w.z).project(camera);
        return { u: p.x * 0.5 + 0.5, v: p.y * 0.5 + 0.5, progress: d.progress(w, now), strength: p.z < 1 ? w.strength : 0 };
      }),
    );
    const [sx, sy] = d.shake(now);
    shockwave.setShake(sx, sy);

    const mesh = ripplesRef.current;
    if (!mesh) return;
    for (let i = 0; i < MAX_RIPPLES; i++) {
      const r = d.ripples[i];
      if (!r) {
        mesh.setMatrixAt(i, scratch.m.makeScale(0, 0, 0));
        continue;
      }
      const t = d.progress(r, now);
      const radius = 1 + r.radius * (1 - (1 - t) * (1 - t));
      scratch.m.makeScale(radius, 1, radius).setPosition(r.x, 0.12, r.z);
      mesh.setMatrixAt(i, scratch.m);
      const fade = (1 - t) * 2.2;
      mesh.setColorAt(i, scratch.c.setRGB(r.rgb[0] * fade, r.rgb[1] * fade, r.rgb[2] * fade));
    }
    mesh.instanceMatrix.needsUpdate = true;
    if (mesh.instanceColor) mesh.instanceColor.needsUpdate = true;
  });

  return (
    <instancedMesh ref={ripplesRef} args={[undefined, undefined, MAX_RIPPLES]} frustumCulled={false}>
      <ringGeometry args={[0.92, 1, 64]} onUpdate={(g) => g.rotateX(-Math.PI / 2)} />
      <meshBasicMaterial toneMapped={false} transparent depthWrite={false} blending={THREE.AdditiveBlending} />
    </instancedMesh>
  );
}
```
`ringGeometry`'s `onUpdate` must rotate only once. If lint or behaviour objects, build the rotated ring geometry in a `useMemo` and pass it with `geometry={…}`, disposing it in an effect.

// Builds the city layout off the main thread (≈ 0.8 s on London). One request, one response.
const scope = self as unknown as {
  onmessage: ((e: MessageEvent<{ world: World }>) => void) | null;
  postMessage(message: unknown): void;
};
scope.onmessage = (e) => {
  try {
    scope.postMessage({ ok: true, layout: buildLayout(e.data.world) });
  } catch (err) {
    scope.postMessage({ ok: false, error: err instanceof Error ? err.message : String(err) });
  }
};

function layoutInWorker(world: World): Promise<CityLayout> {
  return new Promise((resolve, reject) => {
    const worker = new Worker(new URL("../render/layout.worker.ts", import.meta.url), { type: "module" });
    worker.onmessage = (e: MessageEvent<{ ok: boolean; layout?: CityLayout; error?: string }>) => {
